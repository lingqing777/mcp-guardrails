// ~/.cursor/mcp.json { "mcpServers": { "guardrails": { "url": "http://localhost:4000" } } }

# 原配置 REST_BASE_URL=http://localhost:3000 # 改为 ↓ REST_BASE_URL=http://localhost:8081
